#!/usr/bin/python3 -I
# Unabhaengige Kontrastmessung am FERTIGEN JPG (nicht am Hintergrund vor dem Text): «.ch»-Box laut Bauprotokoll, Glyphenkern = Pixel nahe der Akzentfarbe,
# Hintergrund = Pixel weit von Akzentfarbe UND von der Textfarbe; WCAG-Kontrast Akzent gegen schlechtestes / Median-Hintergrundpixel.
import numpy as np
from PIL import Image
def lin(c):
    c = c / 255.0
    return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)
def lum(rgb): return 0.2126 * lin(rgb[..., 0]) + 0.7152 * lin(rgb[..., 1]) + 0.0722 * lin(rgb[..., 2])
FAELLE = [  # Datei, Box x0,x1,y0,y1 (aus dem Bauprotokoll), Akzentfarbe, Textfarbe
    ("out/story-1-wasser-speaker.jpg", (724, 798, 1512, 1549), (201, 171, 124), (243, 238, 231)),
    ("out/story-3-rose-led.jpg",       (724, 798, 1480, 1517), (95, 74, 48),   (48, 42, 37)),
    ("out/story-4-anti-gravitation.jpg",(724, 798, 1532, 1569), (95, 74, 48),   (48, 42, 37)),
    ("out/story-5-gemuesehobel.jpg",   (593, 667, 1512, 1549), (95, 74, 48),   (48, 42, 37)),
    ("out/story-6-kaffeebecher.jpg",   (724, 798, 1480, 1517), (222, 192, 146), (243, 238, 231)),
    ("out_runde1/story-3-rose-led.jpg",(724, 798, 1496, 1533), (139, 115, 85), (48, 42, 37)),
]
def dilate(m, r):
    out = m.copy()
    for _ in range(r):
        o = out.copy()
        o[1:, :] |= out[:-1, :]; o[:-1, :] |= out[1:, :]; o[:, 1:] |= out[:, :-1]; o[:, :-1] |= out[:, 1:]
        o[1:, 1:] |= out[:-1, :-1]; o[:-1, :-1] |= out[1:, 1:]; o[1:, :-1] |= out[:-1, 1:]; o[:-1, 1:] |= out[1:, :-1]
        out = o
    return out
for f, (x0, x1, y0, y1), acc, txt in FAELLE:
    reg = np.asarray(Image.open(f).convert("RGB")).astype(float)[y0:y1, x0:x1]
    d_acc = np.linalg.norm(reg - np.array(acc), axis=2)
    # Glyphe + Kantenpixel: alles, was dem Hintergrund nicht gleicht. Hintergrund-Schaetzung = Median der Randzeilen der Box (oben/unten 3 Zeilen, ausserhalb der Glyphen)
    ring = np.concatenate([reg[:3].reshape(-1, 3), reg[-3:].reshape(-1, 3)])
    bgmed = np.median(ring, axis=0)
    dist_bg = np.linalg.norm(reg - bgmed, axis=2)
    ink = dilate(dist_bg > 25, 3)            # Glyphe, Kanten, Ringing: 3 px Sicherheitssaum
    bgmask = ~ink
    core = d_acc < 35
    la = float(lum(np.array(acc, float)))
    lb = lum(reg[bgmask])
    cr = (np.maximum(lb, la) + 0.05) / (np.minimum(lb, la) + 0.05)
    kern = np.median(reg[core], axis=0) if core.sum() else None
    print("%-36s Kern-Median %s | Hintergrundpixel %5d (Median %s) | Kontrast schlechtestes %.2f, Median %.2f" % (f, None if kern is None else tuple(int(v) for v in kern), bgmask.sum(), tuple(int(v) for v in bgmed), cr.min(), np.median(cr)))
