def frequency(note):
    return 440 * 2 ** ((note - 69) / 12)

def pan(x, position=0):
    angle = (position + 1) * np.pi / 4
    return np.column_stack((x * np.cos(angle), x * np.sin(angle)))

def add(stem, x, start, gain=1, position=0):

    if i >= N: return
    if x.ndim == 1: x = pan(x, position)
    left = max(0, i)
    source_start = max(0, -i)
    count = min(len(x) - source_start, N - left)

def soft_key(note, duration=2.8, touch=1):
    t = np.arange(round(duration * SR)) / SR
    f = frequency(note)
    # Soft, tine-like piano from oscillators. No soundfont or recorded instrument.
    attack = 1 - np.exp(-t / 0.009)
    body = np.exp(-t / (1.25 + (72 - note) * 0.012))

    fm = 0.42 * np.exp(-t / 0.11) * np.sin(phase * 2)
